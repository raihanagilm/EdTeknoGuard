PENJELASAN ARSITEKTUR NOTIFIKASI NATIVE (BUKAN DARI WEB BROWSER):
========================================================================================
Aplikasi Android dan iOS yang dibuat BUKAN sekadar menampilkan notifikasi alert/pop-up di dalam browser web.
Sistem ini menggunakan DUA LAYER NOTIFIKASI NATIVE MURNI DARI SISTEM OPERASI HP:

1. LAYER BACKGROUND SERVICE (Saat Aplikasi Sedang Ditutup / HP Standby di Kantong):
   - Android: `EdTeknoBackgroundService` (Foreground Service + WakeLock + Alarm Tone).
   - iOS: `BGAppRefreshTask` / Background URLSession Polling.
   - Hasil: Bunyi alarm sirine darurat berdering dari status bar sistem Android/iPhone saat redaman drop <= -27 dBm atau ada tiket gangguan baru, meskipun aplikasi TIDAK sedang dibuka.

2. LAYER JAVASCRIPT INTERFACE / BRIDGE (Saat Aplikasi Sedang Dibuka di Layar):
   - Saat user membuka aplikasi dan web ingin memberi notifikasi (misal: "Tiket terkirim", "Data tersimpan", "Peringatan"), web memanggil fungsi bridge:
     `window.AndroidBridge.showNativeNotification(title, message, isEmergency)` di Android, atau
     `window.webkit.messageHandlers.iosBridge.postMessage(...)` di iOS.
   - Hasil: Muncul push notifikasi native resmi dari sistem operasi (bukan sekadar toast/modal web HTML).

        1. `showNativeNotification`: Memunculkan Heads-up banner melayang dari sistem OS.

        ```kotlin
        class WebAppInterface(private val activity: Activity, private val webView: WebView) {
            
            // 1. Heads-Up Notification (Melayang di atas layar HP)
            @JavascriptInterface
            fun showNativeNotification(title: String, message: String, isEmergency: Boolean) {
                val channelId = if (isEmergency) EdTeknoBackgroundService.CHANNEL_EMERGENCY_ID else EdTeknoBackgroundService.CHANNEL_CUSTOMER_ID
                val notification = NotificationCompat.Builder(activity, channelId)
                    .setContentTitle(title)
                    .setContentText(message)
                    .setStyle(NotificationCompat.BigTextStyle().bigText(message))
                    .setSmallIcon(R.mipmap.ic_launcher)
                    .setAutoCancel(true)
                    .setPriority(NotificationCompat.PRIORITY_MAX)
                    .setDefaults(NotificationCompat.DEFAULT_ALL)
                    .setVisibility(NotificationCompat.VISIBILITY_PUBLIC)
                    .build()
                val manager = activity.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
                manager.notify(System.currentTimeMillis().toInt(), notification)
            }

            // 2. Pop-up Notifikasi In-App (Jendela/Kotak Pop-up di dalam layar HP)
            @JavascriptInterface
            fun showInAppPopup(title: String, message: String) {
                activity.runOnUiThread {
                    AlertDialog.Builder(activity)
                        .setTitle(title)
                        .setMessage(message)
                        .setIcon(R.mipmap.ic_launcher)
                        .setPositiveButton("Tutup", null)
                        .show()
                }
            }

            // 3. Dialog Box / Kotak Konfirmasi (Pilihan Aksi Yes / No atau Batal / Lanjut)
            @JavascriptInterface
            fun showConfirmDialog(title: String, message: String, callbackId: String) {
                activity.runOnUiThread {
                    AlertDialog.Builder(activity)
                        .setTitle(title)
                        .setMessage(message)
                        .setIcon(R.mipmap.ic_launcher)
                        .setPositiveButton("Ya / Lanjutkan") { _, _ ->
                            webView.evaluateJavascript("window.onDialogResult('$callbackId', true)", null)
                        }
                        .setNegativeButton("Batal") { _, _ ->
                            webView.evaluateJavascript("window.onDialogResult('$callbackId', false)", null)
                        }
                        .setCancelable(false)
                        .show()
                }
            }
        }
        ```

4. KODE LENGKAP SWIFT (`ViewController.swift`):
   ```swift
   import UIKit
   import WebKit
   import UserNotifications

   class ViewController: UIViewController, WKNavigationDelegate, WKUIDelegate, WKScriptMessageHandler {
       
       var webView: WKWebView!

       override func viewDidLoad() {
           super.viewDidLoad()
           
           // 1. Minta Izin Notifikasi Native iOS saat App Pertama Dibuka
           UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .badge, .sound]) { granted, error in
               if granted {
                   print("Izin notifikasi native iOS diberikan")
               }
           }
           
           // 2. Setup WKWebView Configuration & JavaScript Bridge
           let contentController = WKUserContentController()
           contentController.add(self, name: "iosBridge")
           
           let config = WKWebViewConfiguration()
           config.userContentController = contentController
           config.allowsInlineMediaPlayback = true
           
           webView = WKWebView(frame: view.bounds, configuration: config)
           webView.autoresizingMask = [.flexibleWidth, .flexibleHeight]
           webView.navigationDelegate = self
           webView.uiDelegate = self
           view.addSubview(webView)
           
           // 3. Tentukan URL berdasarkan Target / Scheme
           #if TARGET_GUARD
           let initialUrl = "http://172.21.20.62:7890/login"
           #else
           let initialUrl = "http://172.21.20.62:7890/portal"
           #endif
           
           if let url = URL(string: initialUrl) {
               let request = URLRequest(url: url)
               webView.load(request)
           }
           
           // 4. Polling Latar Belakang (Timer 30 Detik) untuk Cek Notifikasi Server
           Timer.scheduledTimer(withTimeInterval: 30.0, repeats: true) { [weak self] _ in
               self?.pollServerNotification()
           }
       }

       // MARK: - JavaScript Bridge (Dipanggil saat Web mentrigger Notifikasi Native)
       func userContentController(_ userContentController: WKUserContentController, didReceive message: WKScriptMessage) {
           if message.name == "iosBridge", let body = message.body as? [String: Any] {
               let title = body["title"] as? String ?? "Pemberitahuan"
               let msg = body["message"] as? String ?? ""
               let isEmergency = body["isEmergency"] as? Boolean ?? false
               showNativeIOSNotification(title: title, message: msg, isEmergency: isEmergency)
           }
       }

       // MARK: - Eksekusi Notifikasi Native Sistem Operasi iOS
       func showNativeIOSNotification(title: String, message: String, isEmergency: Bool) {
           let content = UNMutableNotificationContent()
           content.title = title
           content.body = message
           content.sound = isEmergency ? UNNotificationSound.defaultCritical : UNNotificationSound.default
           
           // Trigger seketika (1 detik)
           let trigger = UNTimeIntervalNotificationTrigger(timeInterval: 1, repeats: false)
           let request = UNNotificationRequest(identifier: UUID().uuidString, content: content, trigger: trigger)
           
           UNUserNotificationCenter.current().add(request) { error in
               if let error = error {
                   print("Gagal menampilkan notifikasi iOS: \(error)")
               }
           }
       }

       // MARK: - Polling Status Latar Belakang ke Backend Lokal
       func pollServerNotification() {
           guard let url = URL(string: "http://172.21.20.62:7890/api/notifications/poll") else { return }
           
           URLSession.shared.dataTask(with: url) { [weak self] data, response, error in
               guard let data = data, error == nil else { return }
               do {
                   if let json = try JSONSerialization.jsonObject(with: data) as? [String: Any],
                      let unread = json["unread_tickets_count"] as? Int, unread > 0,
                      let latest = json["latest_ticket"] as? [String: Any] {
                       
                       let ticketId = latest["id_tiket"] as? String ?? "TIKET-BARU"
                       let cust = latest["nama_pelanggan"] as? String ?? "Pelanggan"
                       let desc = latest["deskripsi"] as? String ?? ""
                       
                       DispatchQueue.main.async {
                           self?.showNativeIOSNotification(
                               title: "🚨 NOTIFIKASI TIKET NOC",
                               message: "[\(ticketId)] \(cust): \(desc)",
                               isEmergency: true
                           )
                       }
                   }
               } catch { }
           }.resume()
       }
   }
   ```
